import {
  Image as ImageIcon,
  Mic,
  Send,
} from "lucide-react-native";
import React, { useState } from "react";
import {
  Keyboard,
  Platform,
  Pressable,
  StyleSheet,
  TextInput,
  View,
} from "react-native";

type Props = {
  value: string;
  onChange: (text: string) => void;
  onSend: () => void;
  onTyping?: (typing: boolean) => void;
  onSendImage: () => void;       // 🔑 CHỈ LÀ CALLBACK
  onRecordVoice: (file: any) => void;
};

export default function InputBar({
  value,
  onChange,
  onSend,
  onTyping,
  onSendImage,
  onRecordVoice,
}: Props) {
  const [focused, setFocused] = useState(false);

  const handleChangeText = (text: string) => {
    onChange(text);
    onTyping?.(text.length > 0);
  };

  const handleSendPress = () => {
    if (!value.trim()) return;
    Keyboard.dismiss();
    onSend();
    onTyping?.(false);
  };

  return (
    <View style={styles.container}>
      {/* IMAGE PICK */}
      <Pressable
        style={styles.iconBtn}
        onPress={onSendImage}   // ✅ KHÔNG () — CỰC KỲ QUAN TRỌNG
      >
        <ImageIcon size={22} color="#555" />
      </Pressable>

      {/* INPUT */}
      <TextInput
        value={value}
        onChangeText={handleChangeText}
        placeholder="Nhập tin nhắn..."
        placeholderTextColor="#999"
        multiline
        style={[
          styles.input,
          focused && { borderColor: "#007AFF" },
        ]}
        onFocus={() => setFocused(true)}
        onBlur={() => setFocused(false)}
      />

      {/* SEND / MIC */}
      {value.trim() ? (
        <Pressable style={styles.sendBtn} onPress={handleSendPress}>
          <Send size={20} color="#fff" />
        </Pressable>
      ) : (
        <Pressable
          style={styles.iconBtn}
          onPress={() => {
            // TODO: mở record audio → trả file
            // onRecordVoice(file)
          }}
        >
          <Mic size={22} color="#555" />
        </Pressable>
      )}
    </View>
  );
}
